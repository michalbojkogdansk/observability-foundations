Plain text — custom regex per query
# Every query needs hand-crafted regex.
# Example: find duration_ms > 500
import re
pattern = re.compile(
    r'(?:\[(\d+)ms\]'
    r'|\b(\d+)ms\b'
    r'|took\s+(\d+))'
)
results = []
for line in log_lines:
    m = pattern.search(line)
    if m:
        val = int(next(
            g for g in m.groups() if g
        ))
        if val > 500:
            results.append(line)
# Risk: "attempts=3" or HTTP status
# codes can produce false positives.
# Regex breaks on message rewording.
key=value — consistent field parsing
# Split on spaces, extract key=value pairs
results = []
for line in log_lines:
    fields = {}
    for token in line.split():
        if '=' in token:
            k, v = token.split('=', 1)
            fields[k] = v
    duration = int(
        fields.get('duration_ms', 0)
    )
    if duration > 500:
        results.append(line)
# Reliable: field names are explicit.
# Template-independent: message text
# changes never break queries.
# Still O(n): parse all lines.
JSON — native parse, typed fields
import json

results = []
for line in log_lines:
    obj = json.loads(line)
    # duration_ms is always an int —
    # never confused with other fields
    if obj.get('duration_ms', 0) > 500:
        results.append(obj)

# Fast: C-extension JSON parser.
# Safe: no regex, zero ambiguity.
# Indexable: add ElasticSearch or
# DuckDB for sub-ms multi-field
# queries across billions of lines.
